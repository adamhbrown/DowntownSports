    <script>
        // Exit Tunnel Locker Room Scene trigger
        function exitLockerRoom() {
            const screen = document.getElementById('stadium-entrance');
            screen.style.opacity = '0';
            setTimeout(() => { screen.style.visibility = 'hidden'; }, 1500);
        }

        // Target Global Real Manager Identity Setup Arrays
        const networkManagers = [
            "Evan Mandel", "Adam Brown", "Manager 3", "Manager 4", 
            "Manager 5", "Manager 6", "Manager 7", "Manager 8"
        ];

        const baseLeagueTeams = [
            "Mexico", "South Korea", "South Africa", "Czech Republic",
            "Switzerland", "Canada", "Qatar", "Bosnia & Herz.",
            "Brazil", "Morocco", "Scotland", "Haiti",
            "USA", "Australia", "Paraguay", "Turkey"
        ].sort();

        // Build Multi-Round Linear Snake Sequences Array (6 Rounds * 8 Managers = 48 Total Picks)
        const totalRounds = 6;
        let completeDraftOrder = [];

        for (let r = 1; r <= totalRounds; r++) {
            let iterationRoundList = [...networkManagers];
            if (r % 2 === 0) { iterationRoundList.reverse(); } // Snake Twist
            completeDraftOrder.push(...iterationRoundList);
        }

        // Memory Cache State Load Tracking Hooks
        let currentPickIndex = 0;
        let draftRegistry = {}; // Format: { pickIndex: { manager: X, team: Y } }
        let teamPerformanceData = {}; // Format: { teamName: { w: X, l: Y, pts: Z } }

        // Core Init Setup Function
        window.onload = function() {
            if (localStorage.getItem('downtownDraftRegistry')) {
                draftRegistry = JSON.parse(localStorage.getItem('downtownDraftRegistry'));
                currentPickIndex = Object.keys(draftRegistry).length;
            }
            if (localStorage.getItem('downtownPerformanceCache')) {
                teamPerformanceData = JSON.parse(localStorage.getItem('downtownPerformanceCache'));
            } else {
                baseLeagueTeams.forEach(t => { teamPerformanceData[t] = { w: 0, l: 0, pts: 0 }; });
            }

            refreshDraftSelectors();
            renderDraftBoardGrid();
            syncStandingsUI();
            calculateLeaderboard();
        };

        function refreshDraftSelectors() {
            const teamDropdown = document.getElementById('team-select');
            const scoreDropdown = document.getElementById('score-team-select');
            
            teamDropdown.innerHTML = '';
            scoreDropdown.innerHTML = '';

            const selectedTeams = Object.values(draftRegistry).map(obj => obj.team);

            baseLeagueTeams.forEach(team => {
                // Populate interactive draft list with remaining open teams
                if (!selectedTeams.includes(team)) {
                    let opt = document.createElement('option');
                    opt.value = team; opt.textContent = team;
                    teamDropdown.appendChild(opt);
                }
                // Admin Scoring Control list gets all teams
                let optScore = document.createElement('option');
                optScore.value = team; optScore.textContent = team;
                scoreDropdown.appendChild(optScore);
            });
        }

        function renderDraftBoardGrid() {
            for (let i = 1; i <= totalRounds; i++) {
                document.getElementById(`round${i}-list`).innerHTML = '';
            }

            completeDraftOrder.forEach((manager, idx) => {
                const roundIndexNum = Math.floor(idx / networkManagers.length) + 1;
                const parentList = document.getElementById(`round${roundIndexNum}-list`);
                
                let li = document.createElement('li');
                if (draftRegistry[idx]) {
                    li.innerHTML = `<strong>${draftRegistry[idx].manager}:</strong> <span style="color:#ffcc00;">${draftRegistry[idx].team}</span>`;
                } else {
                    li.innerHTML = `<span style="opacity:0.25;">${manager}</span>`;
                }
                parentList.appendChild(li);
            });

            // Update Header Tracking Output Displays
            const statusBox = document.getElementById('draft-status');
            const managerDropdown = document.getElementById('manager-select');

            if (currentPickIndex >= completeDraftOrder.length) {
                statusBox.textContent = "🏆 DRAFT CONCLUDED - BOARD IS LOCKED.";
                document.getElementById('draft-btn').disabled = true;
            } else {
                const nextActiveManager = completeDraftOrder[currentPickIndex];
                const activeRound = Math.floor(currentPickIndex / networkManagers.length) + 1;
                const activePick = (currentPickIndex % networkManagers.length) + 1;
                
                statusBox.textContent = `⚡ PICK WINDOW: Round ${activeRound}, Pick ${activePick} ➔ ${nextActiveManager} is on the clock!`;
                managerDropdown.value = nextActiveManager;
            }
        }

        function executeDraftPick() {
            const teamDropdown = document.getElementById('team-select');
            const targetTeam = teamDropdown.value;
            if (!targetTeam) return;

            const activeManager = completeDraftOrder[currentPickIndex];
            
            draftRegistry[currentPickIndex] = { manager: activeManager, team: targetTeam };
            currentPickIndex++;

            localStorage.setItem('downtownDraftRegistry', JSON.stringify(draftRegistry));
            
            refreshDraftSelectors();
            renderDraftBoardGrid();
            calculateLeaderboard();
        }

        function submitMatchScore() {
            const targetTeam = document.getElementById('score-team-select').value;
            const outcomeRule = document.getElementById('score-outcome-select').value;

            if (outcomeRule === "reg-win") {
                teamPerformanceData[targetTeam].w += 1;
                teamPerformanceData[targetTeam].pts += 3;
            } else if (outcomeRule === "ot-win") {
                teamPerformanceData[targetTeam].w += 1;
                teamPerformanceData[targetTeam].pts += 2;
            } else if (outcomeRule === "so-win") {
                teamPerformanceData[targetTeam].w += 1;
                teamPerformanceData[targetTeam].pts += 1;
            } else if (outcomeRule === "loss") {
                teamPerformanceData[targetTeam].l += 1;
            }

            localStorage.setItem('downtownPerformanceCache', JSON.stringify(teamPerformanceData));
            syncStandingsUI();
            calculateLeaderboard();
        }

        function syncStandingsUI() {
            Object.keys(teamPerformanceData).forEach(team => {
                const tr = document.querySelector(`tr[data-team="${team}"]`);
                if (tr) {
                    tr.querySelector('.w').textContent = teamPerformanceData[team].w;
                    tr.querySelector('.l').textContent = teamPerformanceData[team].l;
                    tr.querySelector('.pts').textContent = teamPerformanceData[team].pts;
                }
            });
            
            // Auto Sort Group Matrices Logic based on updated performance point calculations
            ["A", "B", "C", "D"].forEach(letter => {
                const tbody = document.getElementById(`group-${letter}-rows`);
                let structuralRows = Array.from(tbody.querySelectorAll('tr'));
                
                structuralRows.sort((x, y) => {
                    return parseInt(y.querySelector('.pts').textContent) - parseInt(x.querySelector('.pts').textContent);
                });
                structuralRows.forEach(row => tbody.appendChild(row));
            });
        }

        function calculateLeaderboard() {
            let managerPointMatrix = {};
            networkManagers.forEach(m => managerPointMatrix[m] = 0);

            // Connect drafted teams directly to manager scorecard point systems
            Object.values(draftRegistry).forEach(pick => {
                if (teamPerformanceData[pick.team]) {
                    managerPointMatrix[pick.manager] += teamPerformanceData[pick.team].pts;
                }
            });

            // Format data into table objects
            let sortingBoard = Object.keys(managerPointMatrix).map(m => {
                return { name: m, points: managerPointMatrix[m] };
            });

            sortingBoard.sort((x, y) => y.points - x.points);

            const displayTableBody = document.getElementById('leaderboard-body');
            displayTableBody.innerHTML = '';

            sortingBoard.forEach((managerObj, rankIdx) => {
                let tr = document.createElement('tr');
                tr.innerHTML = `<td>${rankIdx + 1}</td><td>${managerObj.name}</td><td><strong>${managerObj.points}</strong></td>`;
                displayTableBody.appendChild(tr);
            });
        }

        function clearDraftCache() {
            if (confirm("Reset layout? All drafted configurations will clear out completely.")) {
                localStorage.removeItem('downtownDraftRegistry');
                localStorage.removeItem('downtownPerformanceCache');
                location.reload();
            }
        }
    </script>
